💻 Примеры использования для программиста

1. Базовое подключение сканера:

// Создание экземпляра сканера const scanner = new BarcodeScanner({ onScan: (code) => { console.log('Отсканирован код:', code); // Здесь обрабатываем отсканированный код }, onConnect: () => { console.log('Сканер подключен'); }, onDisconnect: () => { console.log('Сканер отключен'); }, onError: (message, error) => { console.error('Ошибка сканера:', message, error); }, terminator: '\t', // Символ окончания кода baudRate: 9600, // Скорость передачи debug: false // Отладочные сообщения }); // Подключение к сканеру await scanner.connect();

2. Работа с сохраненными портами:

// Загрузка сохраненного порта const savedPortInfo = await scanner.loadSavedPortInfo(); if (savedPortInfo) { // Подключение к сохраненному порту await scanner.connect({ trySavedPortFirst: true, savedPortInfo: savedPortInfo }); } else { // Обычное подключение с выбором порта await scanner.connect(); }

3. Управление буфером данных:

// Очистка буфера scanner.clearBuffer(); // Получение статистики буфера const stats = scanner.getBufferStats(); console.log('Размер буфера:', stats.bufferSize); console.log('Использование:', stats.bufferUsagePercent + '%'); // Проверка подключения if (scanner.isScannerConnected()) { console.log('Сканер подключен'); }

4. Отключение сканера:

// Отключение сканера await scanner.disconnect();
